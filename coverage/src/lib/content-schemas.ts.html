export const profileSchema = z.object({
  name: z.string().min(1),
  wordmark: z.string().min(1),
  title: z.string().min(1),
  headline: z.array(z.string().min(1)).min(1),
  tagline: z.string().min(1),
  intro: z.string().min(1),
  availability: z.object({ available: z.boolean(), label: z.string().min(1) }),
  email: z.string().email(),
  location: z.string().optional(),
  heroPortrait: imageRef,
  seo: z.object({ title: z.string().min(1), description: z.string().min(1) }),
  about: z.object({
    lede: z.string().min(1),
    paragraphs: z.array(z.string().min(1)).min(1),
    portrait: imageRef,
    highlights: z.array(z.object({ title: z.string().min(1), text: z.string().min(1) })),
    philosophy: z.string().min(1),
    stats: z.array(z.object({ value: z.string().min(1), label: z.string().min(1) })).default([]),
  }),
  contact: z.object({
    heading: z.string().min(1),

    modalTitle: z.string().min(1),
    modalDescription: z.string().min(1),
  }),
})
export type Profile = z.infer<typeof profileSchema>
 
export const experienceSchema = z.object({
  items: z
    .array(
      z.object({
        id: z.string().min(1),
        company: z.string().min(1),
        position: z.string().min(1),
        period: z.string().min(1),
        logo: z.string().optional(),
        description: z.string().default(''),
        responsibilities: z.array(z.string()).default([]),
        achievements: z.array(z.string()).default([]),
        technologies: z.array(z.string()).default([]),
        placeholder: z.boolean().default(false),
      }),

    .min(1),
})
export type ExperienceContent = z.infer<typeof experienceSchema>
export type ExperienceItem = ExperienceContent['items'][number]

export function formatZodError(name: string, error: z.ZodError): string {
  const lines = error.issues.map((i) => `  - ${i.path.join('.') || '(root)'}: ${i.message}`)
  return `Invalid content in src/content/${name}.yaml\n${lines.join('\n')}`
}
